% The cOS logo. Build with ./build.sh, which writes logo-light.svg and logo-dark.svg.
\documentclass[border=10pt, multi=logo]{standalone}

\usepackage{fontspec}
\setmainfont{Optima}

\usepackage{xcolor}
\usepackage{tikz}

% \cosLogo{letters}{c and tagline} draws the wordmark in the given colours.
\newcommand{\cosLogo}[2]{%
  \begin{tikzpicture}
    \node[inner sep=0, anchor=base west] (w) at (0,0)
      {\fontsize{64}{64}\selectfont{\color{#2}c}{\color{#1}\textbf{OS}}};
    \node[anchor=north west, inner sep=0, font=\large\addfontfeature{LetterSpace=12}] (t)
      at ([yshift=-0.55cm]w.south west) {\color{#2}AN OPERATING SYSTEM IN MODERN C++};
    \draw[line width=1.2pt, #1] ([yshift=-0.25cm]w.south west) -- ([yshift=-0.25cm]w.south west -| t.east);
  \end{tikzpicture}}

\newenvironment{logo}{}{}

\begin{document}

\definecolor{light-letters}{RGB}{0,0,0}
\definecolor{light-grey}{RGB}{90,90,90}
\definecolor{dark-letters}{RGB}{240,240,240}
\definecolor{dark-grey}{RGB}{170,170,170}

\begin{logo}\cosLogo{light-letters}{light-grey}\end{logo}

\begin{logo}\cosLogo{dark-letters}{dark-grey}\end{logo}

\end{document}
